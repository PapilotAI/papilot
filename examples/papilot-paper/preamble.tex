  {\newtheorem{theorem}{Theorem}[section]
   \NewDocumentCommand\statement{mmm}{\theoremstyle{#1}\newtheorem{#2}[theorem]{#3}}}

   \declaretheorem[style=plain, numberwithin=section]{theorem}
   \NewDocumentCommand\statement{mmm}{\declaretheorem[style=#1, sibling=theorem, name=#3]{#2}}}
